# Dependencias principales del proyecto

selenium
webdriver-manager # Para la gestión automática de ChromeDriver
requests
openpyxl # Para leer los archivos Excel de Coronel Mayorista
colorama # Para logs coloridos en consola (aunque la GUI es el foco)
tqdm # Para barras de progreso en consola (útil en pruebas de módulos core)
python-dotenv # Para cargar el archivo .env con las credenciales
ttkthemes # Para mejorar la apariencia de la GUI Tkinter (opcional pero recomendado)

# Nota: Tkinter (para la GUI) usualmente viene incluido con la instalación estándar de Python,
# por lo que no necesita listarse aquí.
